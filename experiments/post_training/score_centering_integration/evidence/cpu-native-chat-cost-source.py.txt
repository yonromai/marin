def run(variant):
    start = time.perf_counter()
    if variant == 'legacy':
        dictionaries = [old_flat[i] for i in range(len(rows))]
        display = formatter._create_chat_logprobs(chosen, dictionaries, None, 129, return_as_token_id=True).model_dump()
        formatted = time.perf_counter()
        wire = pickle.dumps(display, protocol=5)
        received = pickle.loads(wire)
        transported = time.perf_counter()
        selected = [select_chat_response_topk(r['top_logprobs'],128) for r in received['content']]
        result = (np.asarray([r[0] for r in selected], dtype=np.int32), np.asarray([r[1] for r in selected], dtype=np.float32), [r['logprob'] for r in received['content']])
    else:
        selected = select_flat_chat_response_topk(old_flat, chosen, 128)
        formatted = time.perf_counter()
        wire = pickle.dumps(selected, protocol=5)
        received = pickle.loads(wire)
        transported = time.perf_counter()
        result = (received.topk_indices, received.topk_logprobs, received.sampled_logprobs)
    end = time.perf_counter()
    return result, {'representation_seconds': formatted-start, 'pickle_roundtrip_seconds': transported-formatted, 'client_seconds': end-transported, 'total_seconds': end-start, 'pickle_bytes': len(wire)}
